abi <abi/3.0>,
#include <tunables/global>

# Reads the NVMe SMART log through the admin interface (SYS_ADMIN) and talks
# to the MQTT broker; writes nothing on the host. s6-overlay and the monitor
# run as root, the web page as user nvmeweb. Measured on Home Assistant OS in
# complain mode before it enforced.
profile nvme_monitor flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  # s6-overlay, the web page's folder and its user; the NVMe admin commands
  capability chown,
  capability dac_override,
  capability fowner,
  capability kill,
  capability setgid,
  capability setuid,
  capability sys_admin,

  network inet dgram,
  network inet6 dgram,
  network inet stream,
  network inet6 stream,
  network unix,
  # glibc's name lookup asks the kernel for the interface addresses
  network netlink raw,

  # s6 stops its services; the Supervisor stops the container
  signal (send, receive),

  # s6-overlay
  /init rix,
  /package/** rix,
  /command/** rix,
  /etc/s6-overlay/** rix,
  /run/** rwk,
  # s6-overlay runs the scripts it copied to /run at the start
  /run/s6/** rix,
  /run/s6-rc*/** rix,
  /run/service/** rix,
  /run/ rw,

  # Programs and libraries of the image
  /bin/** rix,
  /sbin/** rix,
  /usr/bin/** rix,
  /usr/sbin/** rix,
  /usr/local/bin/** rix,
  /usr/lib/ r,
  /usr/lib/** rm,
  /usr/lib/bashio/bashio rix,
  /usr/local/lib/ r,
  /usr/local/lib/** rm,
  /usr/local/share/** r,
  /lib/** rm,
  /usr/share/** r,
  /etc/** r,

  # bashio
  /tmp/ r,
  /tmp/** rwk,

  # Options and the web page's history
  /data/ r,
  /data/** rwk,

  # The drives, read through the admin interface
  /dev/nvme[0-9] rw,
  /sys/class/nvme/ r,
  /sys/class/nvme/** r,
  /sys/devices/** r,

  @{PROC}/@{pid}/** r,
  @{PROC}/ r,
  /dev/null rw,
  /dev/urandom r,
  /dev/random r,
  /dev/tty rw,

  deny /proc/sys/** w,
  deny /sys/** w,
}
